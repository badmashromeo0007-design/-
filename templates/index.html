async function generateScript() { const concept = document.querySelector('textarea').value; const inputs = document.querySelectorAll('input[type="text"]'); const chapter_no = inputs[0] ? inputs[0].value : '1'; const chapter_title = inputs[1] ? inputs[1].value : 'Episode 1'; const system_name = inputs[2] ? inputs[2].value : 'DIVINE SOUL SYSTEM'; const dhruv_rank = document.querySelector('select') ? document.querySelector('select').value : ''; if (!concept) { alert("Kripya Episode Concept likhein!"); return; } const btn = document.querySelector('button'); btn.innerText = "Generating Script... Please Wait ⏳"; try { const response = await fetch('/generate-script', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ concept, chapter_no, chapter_title, system_name, dhruv_rank }) }); const data = await response.json(); btn.innerText = "GENERATE SCRIPT WITH UNIQUE SYSTEM 🎬✨"; if (data.status === 'success') { alert("Script generated successfully!"); console.log(data.script); // Aap is output ko kisi div mein dikha sakte hain } else { alert("Error: " + data.message); } } catch (err) { btn.innerText = "GENERATE SCRIPT WITH UNIQUE SYSTEM 🎬✨"; alert("Failed to fetch: " + err); } }
